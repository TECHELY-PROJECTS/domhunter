'use client'
import { useState, useEffect } from 'react'
import { useRouter } from 'next/navigation'
import { useSession } from 'next-auth/react'
import { Button } from '@/components/ui/button'
import { Card, CardContent } from '@/components/ui/card'
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'

export default function PortfolioPage() {
  const { data: session, status } = useSession()
  const router = useRouter()
  const [watchlist, setWatchlist] = useState<any[]>([])
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    if (status === 'unauthenticated') router.push('/api/auth/signin')
    if (status === 'authenticated') fetchWatchlist()
  }, [status])

  const fetchWatchlist = async () => {
    const res = await fetch('/api/watchlist')
    const data = await res.json()
    setWatchlist(data.items || [])
    setLoading(false)
  }

  const removeFromWatchlist = async (domainId: string) => {
    await fetch(`/api/watchlist?domainId=${domainId}`, { method: 'DELETE' })
    setWatchlist(w => w.filter(i => i.domainId !== domainId))
  }

  if (status === 'loading' || loading) return (
    <div className="flex items-center justify-center min-h-screen">
      <div className="text-gray-500">Loading portfolio...</div>
    </div>
  )

  const totalValue = watchlist.reduce((sum, i) => sum + (i.domain?.metrics?.estimatedValue || 0), 0)
  const buySignals = watchlist.filter(i => i.domain?.metrics?.recommendation === 'BUY').length

  return (
    <div className="max-w-6xl mx-auto py-8 px-4">
      <div className="flex items-center justify-between mb-6">
        <div>
          <h1 className="text-2xl font-bold text-gray-900">My Watchlist</h1>
          <p className="text-gray-500 text-sm mt-1">{watchlist.length} domains tracked</p>
        </div>
        <Button onClick={() => router.push('/explore')} variant="outline">
          + Add Domains
        </Button>
      </div>

      {/* Stats */}
      <div className="grid grid-cols-3 gap-4 mb-6">
        {[
          { label: 'Domains Watched', value: watchlist.length },
          { label: 'Total Est. Value', value: `$${totalValue >= 1000 ? `${(totalValue/1000).toFixed(1)}K` : totalValue}` },
          { label: 'BUY Signals', value: buySignals },
        ].map(s => (
          <Card key={s.label}>
            <CardContent className="pt-4 pb-4 text-center">
              <div className="text-2xl font-bold text-gray-900">{s.value}</div>
              <div className="text-xs text-gray-400 mt-1">{s.label}</div>
            </CardContent>
          </Card>
        ))}
      </div>

      {watchlist.length === 0 ? (
        <div className="text-center py-16 text-gray-400">
          <p className="text-lg mb-3">No domains in watchlist yet</p>
          <Button onClick={() => router.push('/explore')}>Browse Domains →</Button>
        </div>
      ) : (
        <div className="bg-white rounded-xl border border-gray-200 overflow-hidden">
          <Table>
            <TableHeader>
              <TableRow className="bg-gray-50">
                <TableHead>Domain</TableHead>
                <TableHead className="text-center">Score</TableHead>
                <TableHead className="text-center">Brand</TableHead>
                <TableHead className="text-center">DA</TableHead>
                <TableHead className="text-center">Est. Value</TableHead>
                <TableHead className="text-center">Signal</TableHead>
                <TableHead className="text-center">Expires</TableHead>
                <TableHead className="text-center">Added</TableHead>
                <TableHead></TableHead>
              </TableRow>
            </TableHeader>
            <TableBody>
              {watchlist.map((item: any) => {
                const d = item.domain
                const m = d?.metrics
                return (
                  <TableRow key={item.id} className="hover:bg-gray-50 cursor-pointer"
                    onClick={() => router.push(`/domain/${d?.name}`)}>
                    <TableCell className="font-mono font-semibold text-blue-700">{d?.name}</TableCell>
                    <TableCell className="text-center font-bold text-sm">{m?.rarityScore?.toFixed(0) ?? '—'}</TableCell>
                    <TableCell className="text-center text-sm">{m?.brandScore?.toFixed(0) ?? '—'}</TableCell>
                    <TableCell className="text-center text-sm">{m?.domainAuthority?.toFixed(0) ?? '—'}</TableCell>
                    <TableCell className="text-center text-sm font-medium">
                      {m?.estimatedValue ? `$${m.estimatedValue >= 1000 ? `${(m.estimatedValue/1000).toFixed(1)}K` : m.estimatedValue}` : '—'}
                    </TableCell>
                    <TableCell className="text-center">
                      {m?.recommendation && (
                        <span className={`text-xs px-2 py-0.5 rounded font-medium ${
                          m.recommendation === 'BUY' ? 'bg-green-100 text-green-800' :
                          m.recommendation === 'WATCH' ? 'bg-yellow-100 text-yellow-800' :
                          'bg-gray-100 text-gray-500'
                        }`}>{m.recommendation}</span>
                      )}
                    </TableCell>
                    <TableCell className="text-center text-xs text-gray-500">
                      {d?.auctionEndAt ? new Date(d.auctionEndAt).toLocaleDateString() : '—'}
                    </TableCell>
                    <TableCell className="text-center text-xs text-gray-400">
                      {new Date(item.createdAt).toLocaleDateString()}
                    </TableCell>
                    <TableCell onClick={e => e.stopPropagation()}>
                      <button onClick={() => removeFromWatchlist(item.domainId)}
                        className="text-xs text-red-400 hover:text-red-600 px-2 py-1">
                        Remove
                      </button>
                    </TableCell>
                  </TableRow>
                )
              })}
            </TableBody>
          </Table>
        </div>
      )}
    </div>
  )
}

import { NextRequest, NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

export async function GET(req: NextRequest) {
  const session = await getServerSession(authOptions)
  if (!session?.user?.email) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const user = await prisma.user.findUnique({ where: { email: session.user.email } })
  if (!user) return NextResponse.json({ items: [] })

  const items = await prisma.watchlistItem.findMany({
    where: { userId: user.id },
    include: { domain: { include: { metrics: true } } },
    orderBy: { createdAt: 'desc' },
  })

  return NextResponse.json({ items })
}

export async function POST(req: NextRequest) {
  const session = await getServerSession(authOptions)
  if (!session?.user?.email) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { domainName } = await req.json()
  const user = await prisma.user.findUnique({ where: { email: session.user.email } })
  if (!user) return NextResponse.json({ error: 'User not found' }, { status: 404 })

  const domain = await prisma.domain.findUnique({ where: { name: domainName } })
  if (!domain) return NextResponse.json({ error: 'Domain not found' }, { status: 404 })

  const item = await prisma.watchlistItem.upsert({
    where: { userId_domainId: { userId: user.id, domainId: domain.id } },
    create: { userId: user.id, domainId: domain.id },
    update: {},
  })

  return NextResponse.json({ item })
}

export async function DELETE(req: NextRequest) {
  const session = await getServerSession(authOptions)
  if (!session?.user?.email) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const domainId = req.nextUrl.searchParams.get('domainId')
  if (!domainId) return NextResponse.json({ error: 'domainId required' }, { status: 400 })

  const user = await prisma.user.findUnique({ where: { email: session.user.email } })
  if (!user) return NextResponse.json({ error: 'User not found' }, { status: 404 })

  await prisma.watchlistItem.deleteMany({
    where: { userId: user.id, domainId },
  })

  return NextResponse.json({ ok: true })
}
