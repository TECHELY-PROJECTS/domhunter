```
Build the complete /explore page at /src/app/explore/page.tsx.
This is the main DomCop-style domain table. Make it look professional and clean.

'use client'
import { useState, useEffect, useCallback, useTransition } from 'react'
import { useRouter } from 'next/navigation'
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
import { Badge } from '@/components/ui/badge'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Slider } from '@/components/ui/slider'
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'

const TIER_STYLES: Record<string, string> = {
  legendary: 'bg-yellow-400 text-yellow-900 font-bold',
  epic:      'bg-purple-600 text-white',
  rare:      'bg-blue-600 text-white',
  uncommon:  'bg-green-600 text-white',
  common:    'bg-gray-400 text-white',
}

const REC_STYLES: Record<string, string> = {
  BUY:   'bg-green-100 text-green-800 border border-green-300 font-bold',
  WATCH: 'bg-yellow-100 text-yellow-800 border border-yellow-300',
  SKIP:  'bg-gray-100 text-gray-400 border border-gray-200',
}

const NICHE_ICONS: Record<string, string> = {
  tech: '💻', finance: '💰', health: '❤️', ecommerce: '🛒',
  media: '📺', legal: '⚖️', realestate: '🏠', general: '🌐',
}

export default function ExplorePage() {
  const router = useRouter()
  const [isPending, startTransition] = useTransition()
  const [domains, setDomains] = useState<any[]>([])
  const [total, setTotal] = useState(0)
  const [loading, setLoading] = useState(true)
  const [page, setPage] = useState(1)
  const LIMIT = 50

  const [filters, setFilters] = useState({
    q: '',
    minScore: 0,
    minBrandScore: 0,
    minDA: 0,
    minAge: 0,
    minBacklinks: 0,
    tlds: [] as string[],
    niche: '',
    recommendation: '',
    tier: '',
    sortBy: 'rarityScore',
    sortDir: 'desc',
  })

  const fetchDomains = useCallback(async () => {
    setLoading(true)
    try {
      const params = new URLSearchParams({ page: String(page), limit: String(LIMIT) })
      Object.entries(filters).forEach(([k, v]) => {
        if (Array.isArray(v)) { if (v.length) params.set(k, v.join(',')) }
        else if (v !== '' && v !== 0) params.set(k, String(v))
      })
      const res = await fetch(`/api/domains?${params}`)
      const data = await res.json()
      setDomains(data.domains || [])
      setTotal(data.total || 0)
    } finally {
      setLoading(false)
    }
  }, [page, filters])

  useEffect(() => { fetchDomains() }, [fetchDomains])

  const setFilter = (key: string, value: any) => {
    setPage(1)
    setFilters(f => ({ ...f, [key]: value }))
  }

  const toggleTLD = (tld: string) => {
    setFilter('tlds', filters.tlds.includes(tld)
      ? filters.tlds.filter(t => t !== tld)
      : [...filters.tlds, tld])
  }

  const formatNumber = (n?: number) => {
    if (!n) return '—'
    if (n >= 1_000_000) return `${(n/1_000_000).toFixed(1)}M`
    if (n >= 1_000) return `${(n/1_000).toFixed(1)}K`
    return String(n)
  }

  const formatValue = (v?: number) => {
    if (!v) return '—'
    if (v >= 1_000) return `$${(v/1_000).toFixed(1)}K`
    return `$${v}`
  }

  const formatExpiry = (date?: string) => {
    if (!date) return '—'
    const ms = new Date(date).getTime() - Date.now()
    const hours = Math.round(ms / 3_600_000)
    if (hours < 0) return 'Ended'
    if (hours < 24) return <span className="text-red-600 font-bold">{hours}h left</span>
    if (hours < 72) return <span className="text-orange-500">{Math.round(hours/24)}d left</span>
    return new Date(date).toLocaleDateString()
  }

  const totalPages = Math.ceil(total / LIMIT)

  return (
    <div className="flex h-screen bg-gray-50 overflow-hidden">

      {/* ── SIDEBAR ── */}
      <aside className="w-64 bg-white border-r border-gray-200 flex flex-col overflow-y-auto">
        <div className="p-4 border-b border-gray-100">
          <h2 className="font-semibold text-gray-900 text-sm uppercase tracking-wide">Filters</h2>
        </div>

        <div className="p-4 space-y-5 flex-1">

          {/* Keyword search */}
          <div>
            <label className="text-xs font-medium text-gray-500 uppercase tracking-wide">Search</label>
            <Input
              className="mt-1 h-8 text-sm"
              placeholder="keyword in domain..."
              value={filters.q}
              onChange={e => setFilter('q', e.target.value)}
            />
          </div>

          {/* Recommendation */}
          <div>
            <label className="text-xs font-medium text-gray-500 uppercase tracking-wide">Signal</label>
            <div className="mt-1 flex gap-1">
              {['BUY', 'WATCH', 'SKIP'].map(r => (
                <button key={r} onClick={() => setFilter('recommendation', filters.recommendation === r ? '' : r)}
                  className={`flex-1 py-1.5 text-xs rounded font-medium transition-all ${
                    filters.recommendation === r ? REC_STYLES[r] : 'bg-gray-50 text-gray-400 border border-gray-200 hover:border-gray-300'
                  }`}>
                  {r}
                </button>
              ))}
            </div>
          </div>

          {/* Rarity tier */}
          <div>
            <label className="text-xs font-medium text-gray-500 uppercase tracking-wide">Rarity Tier</label>
            <div className="mt-1 flex flex-wrap gap-1">
              {['legendary','epic','rare','uncommon','common'].map(t => (
                <button key={t} onClick={() => setFilter('tier', filters.tier === t ? '' : t)}
                  className={`px-2 py-1 text-xs rounded capitalize transition-all ${
                    filters.tier === t ? TIER_STYLES[t] : 'bg-gray-50 text-gray-500 border border-gray-200'
                  }`}>
                  {t}
                </button>
              ))}
            </div>
          </div>

          {/* TLD */}
          <div>
            <label className="text-xs font-medium text-gray-500 uppercase tracking-wide">TLD</label>
            <div className="mt-1 flex flex-wrap gap-1">
              {['com','io','ai','co','net','org'].map(t => (
                <button key={t} onClick={() => toggleTLD(t)}
                  className={`px-2 py-1 text-xs rounded font-mono transition-all ${
                    filters.tlds.includes(t)
                      ? 'bg-blue-600 text-white'
                      : 'bg-gray-50 text-gray-500 border border-gray-200 hover:border-blue-300'
                  }`}>
                  .{t}
                </button>
              ))}
            </div>
          </div>

          {/* Niche */}
          <div>
            <label className="text-xs font-medium text-gray-500 uppercase tracking-wide">Niche</label>
            <Select value={filters.niche} onValueChange={v => setFilter('niche', v)}>
              <SelectTrigger className="mt-1 h-8 text-sm">
                <SelectValue placeholder="All niches" />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="">All niches</SelectItem>
                {Object.entries(NICHE_ICONS).map(([n, icon]) => (
                  <SelectItem key={n} value={n}>{icon} {n.charAt(0).toUpperCase()+n.slice(1)}</SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>

          {/* Sliders */}
          {[
            { key: 'minScore', label: 'Min Rarity Score', max: 100 },
            { key: 'minBrandScore', label: 'Min Brand Score', max: 100 },
            { key: 'minDA', label: 'Min Domain Authority', max: 80 },
            { key: 'minAge', label: 'Min Age (years)', max: 20 },
          ].map(({ key, label, max }) => (
            <div key={key}>
              <div className="flex justify-between">
                <label className="text-xs font-medium text-gray-500 uppercase tracking-wide">{label}</label>
                <span className="text-xs font-semibold text-gray-700">{filters[key as keyof typeof filters]}</span>
              </div>
              <Slider
                className="mt-2" min={0} max={max} step={5}
                value={[filters[key as keyof typeof filters] as number]}
                onValueChange={([v]) => setFilter(key, v)}
              />
            </div>
          ))}

          {/* Reset */}
          <Button variant="outline" size="sm" className="w-full"
            onClick={() => { setPage(1); setFilters({ q:'', minScore:0, minBrandScore:0, minDA:0, minAge:0, minBacklinks:0, tlds:[], niche:'', recommendation:'', tier:'', sortBy:'rarityScore', sortDir:'desc' }) }}>
            Reset Filters
          </Button>
        </div>
      </aside>

      {/* ── MAIN ── */}
      <main className="flex-1 flex flex-col overflow-hidden">

        {/* Toolbar */}
        <div className="bg-white border-b border-gray-200 px-5 py-3 flex items-center justify-between">
          <p className="text-sm text-gray-600">
            {loading ? 'Loading...' : (
              <>Showing <span className="font-semibold">{((page-1)*LIMIT)+1}–{Math.min(page*LIMIT, total)}</span> of <span className="font-semibold">{total.toLocaleString()}</span> domains</>
            )}
          </p>
          <div className="flex items-center gap-3">
            <Select value={`${filters.sortBy}-${filters.sortDir}`}
              onValueChange={v => { const [f,d] = v.split('-'); setFilters(prev => ({...prev, sortBy:f, sortDir:d})) }}>
              <SelectTrigger className="h-8 w-52 text-sm">
                <SelectValue />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="rarityScore-desc">Rarity Score ↓</SelectItem>
                <SelectItem value="brandScore-desc">Brand Score ↓</SelectItem>
                <SelectItem value="estimatedValue-desc">Est. Value ↓</SelectItem>
                <SelectItem value="domainAuthority-desc">DA ↓</SelectItem>
                <SelectItem value="backlinks-desc">Backlinks ↓</SelectItem>
                <SelectItem value="domainAge-desc">Age ↓</SelectItem>
                <SelectItem value="auctionEndAt-asc">Ending Soon ↑</SelectItem>
              </SelectContent>
            </Select>
          </div>
        </div>

        {/* Table */}
        <div className="flex-1 overflow-auto">
          <Table>
            <TableHeader className="sticky top-0 bg-gray-50 z-10">
              <TableRow>
                <TableHead className="w-52">Domain</TableHead>
                <TableHead className="w-28 text-center">Tier</TableHead>
                <TableHead className="w-20 text-center">Score</TableHead>
                <TableHead className="w-20 text-center">Brand</TableHead>
                <TableHead className="w-16 text-center">DA</TableHead>
                <TableHead className="w-24 text-center">Backlinks</TableHead>
                <TableHead className="w-16 text-center">Age</TableHead>
                <TableHead className="w-24 text-center">Value</TableHead>
                <TableHead className="w-24 text-center">Niche</TableHead>
                <TableHead className="w-20 text-center">Signal</TableHead>
                <TableHead className="w-24 text-center">Expires</TableHead>
                <TableHead className="w-20 text-center">Actions</TableHead>
              </TableRow>
            </TableHeader>
            <TableBody>
              {loading
                ? Array(15).fill(0).map((_, i) => (
                    <TableRow key={i}>
                      {Array(12).fill(0).map((_, j) => (
                        <TableCell key={j}><div className="h-4 bg-gray-100 rounded animate-pulse" /></TableCell>
                      ))}
                    </TableRow>
                  ))
                : domains.map((d: any) => (
                    <TableRow key={d.id} className="hover:bg-blue-50 cursor-pointer group"
                      onClick={() => router.push(`/domain/${d.name}`)}>

                      <TableCell className="font-mono font-semibold text-blue-700 group-hover:text-blue-900">
                        {d.name}
                      </TableCell>

                      <TableCell className="text-center">
                        {d.metrics?.rarityTier && (
                          <span className={`text-xs px-2 py-0.5 rounded-full capitalize ${TIER_STYLES[d.metrics.rarityTier]}`}>
                            {d.metrics.rarityTier}
                          </span>
                        )}
                      </TableCell>

                      <TableCell className="text-center">
                        <span className={`font-bold text-sm ${
                          (d.metrics?.rarityScore||0) >= 70 ? 'text-green-700' :
                          (d.metrics?.rarityScore||0) >= 50 ? 'text-blue-700' : 'text-gray-500'
                        }`}>
                          {d.metrics?.rarityScore?.toFixed(0) ?? '—'}
                        </span>
                      </TableCell>

                      <TableCell className="text-center text-sm text-gray-700">
                        {d.metrics?.brandScore?.toFixed(0) ?? '—'}
                      </TableCell>

                      <TableCell className="text-center">
                        <span className={`text-sm font-medium ${
                          (d.metrics?.domainAuthority||0) >= 50 ? 'text-green-700' :
                          (d.metrics?.domainAuthority||0) >= 20 ? 'text-yellow-700' : 'text-gray-400'
                        }`}>
                          {d.metrics?.domainAuthority?.toFixed(0) ?? '—'}
                        </span>
                      </TableCell>

                      <TableCell className="text-center text-sm text-gray-600">
                        {formatNumber(d.metrics?.backlinks)}
                      </TableCell>

                      <TableCell className="text-center text-sm text-gray-600">
                        {d.metrics?.domainAge ? `${d.metrics.domainAge}yr` : '—'}
                      </TableCell>

                      <TableCell className="text-center text-sm font-semibold text-gray-800">
                        {formatValue(d.metrics?.estimatedValue)}
                      </TableCell>

                      <TableCell className="text-center text-sm">
                        {d.metrics?.niche && (
                          <span title={d.metrics.niche}>
                            {NICHE_ICONS[d.metrics.niche] ?? '🌐'}
                            <span className="ml-1 text-xs text-gray-500 capitalize">{d.metrics.niche}</span>
                          </span>
                        )}
                      </TableCell>

                      <TableCell className="text-center">
                        {d.metrics?.recommendation && (
                          <span className={`text-xs px-2 py-0.5 rounded ${REC_STYLES[d.metrics.recommendation]}`}>
                            {d.metrics.recommendation}
                          </span>
                        )}
                      </TableCell>

                      <TableCell className="text-center text-xs">
                        {formatExpiry(d.auctionEndAt)}
                      </TableCell>

                      <TableCell className="text-center" onClick={e => e.stopPropagation()}>
                        <a
                          href={`https://www.godaddy.com/domainsearch/find?checkAvail=1&domainToCheck=${d.name}`}
                          target="_blank" rel="noopener noreferrer"
                          className="text-xs bg-blue-600 text-white px-2 py-1 rounded hover:bg-blue-700"
                        >
                          Buy →
                        </a>
                      </TableCell>
                    </TableRow>
                  ))
              }
            </TableBody>
          </Table>
        </div>

        {/* Pagination */}
        <div className="bg-white border-t border-gray-200 px-5 py-3 flex items-center justify-between">
          <Button variant="outline" size="sm" disabled={page <= 1} onClick={() => setPage(p => p-1)}>
            ← Previous
          </Button>
          <div className="flex items-center gap-1">
            {Array.from({ length: Math.min(7, totalPages) }, (_, i) => {
              const p = page <= 4 ? i+1 : page - 3 + i
              if (p < 1 || p > totalPages) return null
              return (
                <button key={p} onClick={() => setPage(p)}
                  className={`w-8 h-8 text-sm rounded ${p === page ? 'bg-blue-600 text-white' : 'hover:bg-gray-100 text-gray-600'}`}>
                  {p}
                </button>
              )
            })}
          </div>
          <Button variant="outline" size="sm" disabled={page >= totalPages} onClick={() => setPage(p => p+1)}>
            Next →
          </Button>
        </div>
      </main>
    </div>
  )
}
```
