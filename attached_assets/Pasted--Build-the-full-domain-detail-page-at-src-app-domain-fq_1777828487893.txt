const TIER_COLORS: Record<string, string> = {
  legendary: 'text-yellow-600 bg-yellow-50 border-yellow-200',
  epic:      'text-purple-700 bg-purple-50 border-purple-200',
  rare:      'text-blue-700 bg-blue-50 border-blue-200',

async function getDomain(fqdn: string) {
  const res = await fetch(`${process.env.NEXTAUTH_URL}/api/domains/${fqdn}`, {
    cache: 'no-store'
  })
  if (!res.ok) return null
  return res.json()
}
