async function getStats() {
  const [total, buySignals, withValue] = await Promise.all([
    prisma.domain.count(),
    prisma.metrics.count({ where: { recommendation: 'BUY' } }),
    prisma.metrics.count({ where: { estimatedValue: { gt: 500 } } }),
  ])
  return { total, buySignals, withValue }
}

async function getSampleDomains() {
  return prisma.domain.findMany({
    where: { metrics: { recommendation: 'BUY', rarityScore: { gte: 65 } } },
    include: { metrics: true },
    orderBy: { metrics: { rarityScore: 'desc' } },
    take: 5,
  })
}
